// Rang FIFO pondéré par criticité et ancienneté
FIFO_Queue_Rank = 
RANKX(
    FILTER(ALLSELECTED('LabQueue'), 'LabQueue'[Status] = "Waiting"),
    'LabQueue'[IsUrgent] * 1000 + 'LabQueue'[AgeHours],
    , DESC, DENSE
)

// Taux de disponibilité effective des bancs d'essais
Bench_Availability_Pct = 
DIVIDE(
    SUM('BenchTelemetry'[ActiveRunHours]),
    SUM('BenchTelemetry'[ScheduledShiftHours]) - SUM('BenchTelemetry'[PlannedMaintHours]),
    0
)

# Calcul vectorisé du temps de réponse médian & détection des goulots
df["queue_delay_h"] = (df["start_ts"] - df["arrival_ts"]).dt.total_seconds() / 3600.0
df["sla_breach"] = np.where(
    (df["priority"] == "Urgent") & (df["queue_delay_h"] > 2.0), 1, 0
)

bench_kpi = (
    df.groupby("bench_id", as_index=False)
      .agg(median_resp_min=("resp_min", "median"),
           urgent_backlog=("sla_breach", "sum"))
)
